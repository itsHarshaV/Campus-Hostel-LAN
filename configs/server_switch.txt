! ============================================================
! CAMPUS HOSTEL LAN – SERVER SWITCH CONFIGURATION
! Device: Cisco Catalyst 2960 – hostname: HLN-Server-SW
! Location: Server Room (Basement)
! VLANs: 100 (Servers), 10 (Admin management)
! Servers:
!   192.168.100.2 – DHCP + DNS Server
!   192.168.100.3 – Web/Application Server (hostel.local)
!   192.168.100.4 – Database Server (MySQL)
! ============================================================

hostname HLN-Server-SW
enable secret cisco123
service password-encryption
no ip domain-lookup

! ── VLANs ────────────────────────────────────────────────────
vlan 10
 name ADMIN
vlan 100
 name SERVERS

! ── Uplink to Core Switch – TRUNK ────────────────────────────
interface GigabitEthernet0/1
 description Uplink to HLN-Core-SW Gi0/5
 switchport mode trunk
 switchport trunk allowed vlan 100,10
 no shutdown

! ── DHCP+DNS Server (VLAN 100) ───────────────────────────────
interface FastEthernet0/1
 description DHCP-DNS Server – 192.168.100.2
 switchport mode access
 switchport access vlan 100
 spanning-tree portfast
 no shutdown

! ── Web/App Server (VLAN 100) ────────────────────────────────
interface FastEthernet0/2
 description Web-App Server – 192.168.100.3
 switchport mode access
 switchport access vlan 100
 spanning-tree portfast
 no shutdown

! ── Database Server (VLAN 100) ───────────────────────────────
interface FastEthernet0/3
 description Database Server – 192.168.100.4
 switchport mode access
 switchport access vlan 100
 spanning-tree portfast
 no shutdown

! ── Admin Management Port (VLAN 10) ──────────────────────────
interface FastEthernet0/4
 description Admin Out-of-Band Management
 switchport mode access
 switchport access vlan 10
 spanning-tree portfast
 no shutdown

! ── Port Security on Server Ports ────────────────────────────
interface range FastEthernet0/1 - 4
 switchport port-security
 switchport port-security maximum 1
 switchport port-security violation shutdown
 switchport port-security mac-address sticky

! ── Unused ports – shut down ──────────────────────────────────
interface range FastEthernet0/5 - 24
 description Unused
 switchport mode access
 switchport access vlan 999
 shutdown

! ── Management SVI ───────────────────────────────────────────
interface Vlan100
 ip address 192.168.100.254 255.255.255.0
 no shutdown

ip default-gateway 192.168.100.1

! ── Spanning Tree ────────────────────────────────────────────
spanning-tree mode rapid-pvst

! ── SSH Remote Management ─────────────────────────────────────
ip domain-name hostel.local
crypto key generate rsa modulus 2048
ip ssh version 2
line vty 0 15
 transport input ssh
 login local
 exec-timeout 10 0

username admin privilege 15 secret admin123
line console 0
 login local
 exec-timeout 5 0

end
write memory
