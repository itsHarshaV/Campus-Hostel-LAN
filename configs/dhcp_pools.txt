! ============================================================
! CAMPUS HOSTEL LAN – DHCP POOL CONFIGURATION
! Configured on: 192.168.100.2 (DHCP+DNS Server)
! Or, if using Cisco IOS DHCP, paste onto HLN-Router
! ============================================================

! ── EXCLUDED ADDRESSES (reserved for static devices) ─────────

ip dhcp excluded-address 192.168.10.1  192.168.10.20
ip dhcp excluded-address 192.168.20.1  192.168.20.10
ip dhcp excluded-address 192.168.30.1  192.168.30.10
ip dhcp excluded-address 192.168.40.1  192.168.40.10
ip dhcp excluded-address 192.168.50.1  192.168.50.10
ip dhcp excluded-address 192.168.100.1 192.168.100.20

! ── POOL: VLAN 10 – Hostel Administration ─────────────────────
ip dhcp pool VLAN10_ADMIN
 network 192.168.10.0 255.255.255.0
 default-router 192.168.10.1
 dns-server 192.168.100.2
 domain-name hostel.local
 lease 1 0 0
 ! Range: .21 – .254 (234 addresses)

! ── POOL: VLAN 20 – Students ─────────────────────────────────
ip dhcp pool VLAN20_STUDENTS
 network 192.168.20.0 255.255.255.0
 default-router 192.168.20.1
 dns-server 192.168.100.2
 domain-name hostel.local
 lease 0 8 0
 ! Range: .11 – .254 (244 addresses)
 ! Short lease (8 hours) suits student device churn

! ── POOL: VLAN 30 – Wardens ──────────────────────────────────
ip dhcp pool VLAN30_WARDENS
 network 192.168.30.0 255.255.255.0
 default-router 192.168.30.1
 dns-server 192.168.100.2
 domain-name hostel.local
 lease 1 0 0
 ! Range: .11 – .254

! ── POOL: VLAN 40 – Common / Lab Area ────────────────────────
ip dhcp pool VLAN40_COMMON
 network 192.168.40.0 255.255.255.0
 default-router 192.168.40.1
 dns-server 192.168.100.2
 domain-name hostel.local
 lease 0 4 0
 ! Range: .11 – .254
 ! Short lease (4 hours) for shared lab devices

! ── POOL: VLAN 50 – Guest Network ────────────────────────────
ip dhcp pool VLAN50_GUEST
 network 192.168.50.0 255.255.255.0
 default-router 192.168.50.1
 dns-server 8.8.8.8
 ! Guests use Google DNS – no internal resolution
 lease 0 2 0
 ! Very short lease (2 hours) for guests

! ── NOTE: VLAN 100 Servers use static IPs only ───────────────
! 192.168.100.1  – Router Sub-interface gateway
! 192.168.100.2  – DHCP + DNS Server
! 192.168.100.3  – Web / Application Server (hostel.local)
! 192.168.100.4  – Database Server (MySQL)
! 192.168.100.5  – Reserved (future use)

! ── IP Address Summary Table ──────────────────────────────────
! VLAN │ Network           │ Gateway       │ DHCP Range         │ Static Range
! ─────┼───────────────────┼───────────────┼────────────────────┼──────────────────
!  10  │ 192.168.10.0/24   │ 192.168.10.1  │ .21 – .254         │ .1 – .20
!  20  │ 192.168.20.0/24   │ 192.168.20.1  │ .11 – .254         │ .1 – .10
!  30  │ 192.168.30.0/24   │ 192.168.30.1  │ .11 – .254         │ .1 – .10
!  40  │ 192.168.40.0/24   │ 192.168.40.1  │ .11 – .254         │ .1 – .10
!  50  │ 192.168.50.0/24   │ 192.168.50.1  │ .11 – .254         │ .1 – .10
! 100  │ 192.168.100.0/24  │ 192.168.100.1 │ N/A (static only)  │ .1 – .20
